cmake_minimum_required(VERSION 3.25)
project(ps3_native VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_C_STANDARD 17)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Python3 3.11 REQUIRED COMPONENTS Interpreter)
find_package(Git REQUIRED)
enable_testing()
set(PS3RECOMP_DIR "${CMAKE_CURRENT_SOURCE_DIR}/external/ps3recomp")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/deps.lock.json" lock)
string(JSON pin GET "${lock}" ps3recomp commit)
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${PS3RECOMP_DIR}" rev-parse HEAD
    OUTPUT_VARIABLE actual OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE rc)
if(NOT rc EQUAL 0 OR NOT actual STREQUAL pin)
    message(FATAL_ERROR "Initialize ps3recomp at the deps.lock.json revision: git submodule update --init")
endif()
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/src/ps3native/dependency.py" --apply
    RESULT_VARIABLE rc)
if(NOT rc EQUAL 0)
    message(FATAL_ERROR "ps3recomp adaptation verification failed")
endif()
add_subdirectory(external/ps3recomp EXCLUDE_FROM_ALL)
target_compile_definitions(ps3recomp_runtime PRIVATE PS3_NATIVE_STRICT)
if(MSVC AND CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
    target_compile_options(ps3recomp_runtime PRIVATE /clang:-ffp-contract=off)
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/deps.lock.json")
configure_file(src/launcher_config.h.in generated/launcher_config.h @ONLY)
add_executable(ps3-native src/launcher.cpp)
target_include_directories(ps3-native PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
add_test(NAME catalog COMMAND "${Python3_EXECUTABLE}" -m unittest discover -s tests -p test_*.py)
set_tests_properties(catalog PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}" TIMEOUT 60)
add_subdirectory(tests)
set_tests_properties(catalog PROPERTIES ENVIRONMENT "PS3_NATIVE_FIXTURE_DIR=${CMAKE_CURRENT_BINARY_DIR}/tests/generated")
set(PS3_NATIVE_RECOMP_DIR "" CACHE PATH "Verified local PPU AOT generation")
if(PS3_NATIVE_RECOMP_DIR)
    include(cmake/NativeModule.cmake)
endif()
